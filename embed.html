Fast Integration

Build in minutes, not months

Simple RESTful endpoints with authenticated headers and instant webhook events.

# Create an automated savings plan for your user
curl -X POST https://api.cowrywise.com/v1/savings/plans \
  -H "Authorization: Bearer sec_key_live_948f..." \
  -H "Content-Type: application/json" \
  -d '{
    "customer_id": "cust_847192",
    "plan_name": "My Dream Home",
    "target_amount": 5000000,
    "frequency": "monthly",
    "duration_months": 12
  }'
import { CowrywiseClient } from '@cowrywise/embed-sdk';

const cowry = new CowrywiseClient({ apiKey: process.env.COWRYWISE_SECRET_KEY });

const plan = await cowry.savings.createPlan({
  customerId: 'cust_847192',
  planName: 'My Dream Home',
  targetAmount: 5000000,
  frequency: 'monthly',
  durationMonths: 12
});

console.log('Plan created successfully:', plan.id);
from cowrywise import Client

client = Client(api_key="sec_key_live_948f...")

plan = client.savings.create_plan(
    customer_id="cust_847192",
    plan_name="My Dream Home",
    target_amount=5000000,
    frequency="monthly",
    duration_months=12
)

print(f"Created Plan: {plan.id} with status {plan.status}")
